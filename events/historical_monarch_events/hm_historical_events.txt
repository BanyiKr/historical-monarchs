hm_netherlands.3 = { # Coronation of Queen Wilhelmina
	type = country_event

	title = hm_netherlands.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = hm_netherlands.3.d
				trigger = {
					exists = c:LUX
					c:LUX = {
						is_subject_of = ROOT
						is_subject_type = subject_type_personal_union
					}
				}
			}
			triggered_desc = {
				desc = hm_netherlands.3.d2
				trigger = {
					always = yes # Fallback
				}
			}
		}
	}
	flavor = hm_netherlands.3.f

	gui_window = event_window_1char_tabloid

	left_icon = ruler

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	cooldown = { days = stupidly_long_modifier_time }

	duration = 3

	trigger = {
		# triggered by on_action
	}

	immediate = {
		ruler = {
			save_scope_as = queen_wilhelmina
		}
		if = {
			limit = {
				exists = c:HAN
			}
			c:NET ?= { power_bloc ?= { add_leverage = { target = c:LUX value = -1000 } } }
		}
		c:LUX ?= {
			remove_diplomatic_pact = {
				country = c:NET
				type = personal_union
			}
			if = {
				limit = {
					exists = c:NAS
				}
				c:NAS = {
					create_diplomatic_pact = {
						country = c:LUX
						type = personal_union
					}
				}
			}
			else_if = {
				limit = {
					any_country_in_europe = {
						was_formed_from = NAS
					}
				}
				random_country_in_europe = {
					limit = {
						was_formed_from = NAS
					}
					create_diplomatic_pact = {
						country = c:LUX
						type = personal_union
					}
				}
			}
			else = {
				if = {
					limit = {
						game_date < 1905.11.17
					}
					create_character = {
						template = NAS_adolf_von_nassau_weilburg
						ruler = yes
					}
					create_character = {
						template = LUX_william_iv
						heir = yes
					}
				}
				else_if = {
					limit = {
						game_date < 1912.2.25
					}
					create_character = {
						template = LUX_william_iv
						ruler = yes
					}
					create_character = {
						template = LUX_marie_adelaide_i
						heir = yes
					}
				}
				else_if = {
					limit = {
						game_date < 1924.1.24
					}
					create_character = {
						template = LUX_marie_adelaide_i
						ruler = yes
					}
					create_character = {
						template = LUX_charlotte_i
						heir = yes
					}
				}
				else = {
					create_character = {
						template = LUX_charlotte_i
						ruler = yes
					}
					create_character = {
						template = LUX_jean_i
						heir = yes
					}
				}
			}
			save_scope_as = luxembourg_wilhelmina_scope
		}
		c:NET ?= {
			save_scope_as = netherlands_wilhelmina_country_scope
		}
	}
	option = {
		name = hm_netherlands.3.a # God save the Queen!
		show_as_tooltip = {
			c:LUX ?= {
				remove_diplomatic_pact = {
					country = c:NET
					type = personal_union
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					game_date >= 1896.8.31
					game_date < 1909.4.30
				}
				create_character = {
					template = GER_Heinrich_XXXII_Reuss
					heir = yes
					save_scope_as = heir_scope
				}
			}
			else_if = {
				limit = {
					game_date >= 1909.4.30
				}
				create_character = {
					template = NET_juliana_i
					heir = yes
					save_scope_as = heir_scope
				}
			}
		}
		custom_tooltip = hm_netherlands.3.tt
		show_as_tooltip = {
			scope:queen_wilhelmina = {
				set_character_as_ruler = yes
			}
			if = {
				limit = {
					exists = scope:heir_scope
				}
				set_heir = scope:heir_scope
			}
		}
		hidden_effect = {
			c:LUX ?= {
				trigger_event = {
					id = hm_netherlands.4
					popup = yes
				}
			}
		}
		every_country = {
			limit = {
				has_interest_marker_in_region = sr:region_western_europe
			}
			post_notification = wilhelmina_notification
		}
	}
}

hm_netherlands.5 = { # Death of Prince William

	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 3

	trigger = {
		any_scope_character = {
			has_template = NET_prince_willem
		}
		game_date >= 1879.7.11
		OR = {
			c:NET ?= this
			was_formed_from = NET
		}
	}

	immediate = {
		random_scope_character = {
			limit = {
				is_ruler_of_own_country = yes
			}
			save_scope_as = relevant_ruler
		}
		random_scope_character = {
			limit = {
				has_template = NET_prince_willem
			}
			save_scope_as = relevant_heir
		}
	}

	cancellation_trigger = {
		NOT = {
			any_scope_character = {
				has_template = NET_prince_willem
			}
		}
	}

	option = {
		name = hm_netherlands.5.a
		custom_tooltip = {
			text = relevant_heir_death_tt
			scope:relevant_heir = {
				kill_character = { hidden = yes } # William dies
			}
			create_character = {
				#  Alexander is the new heir
				template = NET_prince_alexander
				heir = yes
			}
		}
		scope:relevant_ruler = {
			add_modifier = {
				name = royal_family_scandal
				days = normal_modifier_time
			}
		}
	}
}

hm_netherlands.6 = { # Abdication of William I
	type = country_event
	placement = ROOT

	title = wedding.3.t
	desc = wedding.3.d1
	flavor = wedding.3.f

	gui_window = event_window_2char

	left_icon = scope:abdicating_monarch
	right_icon = scope:heir_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	cooldown = { days = stupidly_long_modifier_time }

	duration = 3

	trigger = {
		game_date >= 1840.8.24
		OR = {
			c:NET ?= this
			was_formed_from = NET
		}
		ruler ?= {
			has_template = NET_willem_i_van_oranje_nassau
			is_monarch = yes
		}
		calc_true_if = {
			amount >= 2
			
			NOR = {
				has_state_in_state_region = STATE_FLANDERS
				has_state_in_state_region = STATE_WALLONIA
			}
			NOT = {
				has_law_or_variant = law_type:law_autocracy
			}
			is_subject = yes
		}
	}

	immediate = {
		ruler = {
			save_scope_as = abdicating_monarch
			interest_group ?= {
				save_scope_as = monarch_ig
			}
		}
		heir ?= {
			save_scope_as = heir_scope
		}
		law_type:law_monarchy = {
			save_scope_as = monarchy_law
		}
	}

	option = { # reassure the aristocracy
		name = wedding.3.a
		default_option = yes
		add_loyalists = {
			value = 0.1
			pop_type = aristocrats
		}
		scope:heir_scope ?= {
			set_character_as_ruler = yes
		}
		scope:abdicating_monarch ?= {
			resign_and_retire = yes
		}
	}

	option = { # reassure the people
		name = wedding.3.b
		add_loyalists = {
			value = 0.01
		}
		scope:heir_scope ?= {
			set_character_as_ruler = yes
		}
		scope:abdicating_monarch ?= {
			resign_and_retire = yes
		}
	}
}
