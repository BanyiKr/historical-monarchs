GER_character_generate_prussia_federal_monarch = { # Owner = country, ROOT = predecessor
	owner = {
		if = {
			limit = {
				is_german_federal_monarchy = yes
				NOR = {
					was_formed_from = PRU
					exists = c:PRU
				}
			}
			if = {
				limit = {
					s:STATE_BRANDENBURG ?= {
						var:prussia_king_chain_id = 0
					}
					game_date >= 1770.08.03
				}
				create_character = {
					template = pru_wilhelm_III_hohenzollern_template
					role = character_role_magnate
					role = character_role_politician
					holding_type = building_manor_house
					on_created = {
						set_variable = house_hohenzollern
						set_variable = { name = ger_federal_monarch_var value = s:STATE_BRANDENBURG }
