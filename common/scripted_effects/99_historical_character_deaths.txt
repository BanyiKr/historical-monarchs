character_netherlands_scripted_deaths_effect = {

		limit = {
			has_template = NET_willem_i_van_oranje_nassau
			year >= 1840
		}
		kill_character = { hidden = no }
	}
	else_if = {
		limit = {
			has_template = NET_willem_ii_van_oranje_nassau
			year >= 1849
		}
		kill_character = { hidden = no }
	}
	else_if = { # Alexander, Prince of Orange
		limit = {
			has_template = NET_prince_alexander
			year >= 1884
		}
		if = { # Replace heir if still heir
			limit = {
				is_heir_of_own_country = yes
			}
			kill_character = { hidden = no }
			owner ?= {
				create_character = {
					template = NET_wilhelmina_i
					heir = yes
				}
			}
		}
		else = {
			kill_character = { hidden = no }
		}
	}
	else_if = {
		limit = {
			has_template = NET_willem_iii
			year >= 1890
		}
		kill_character = { hidden = no }
	}
}

			}
		}

	}
	else_if = {
		limit = {
			has_template = WAL_gheorghe_bibescu
			year >= 1848
		}
		owner ?= {
			if = {
				limit = {
					has_law = law_type:law_monarchy
					has_law = law_type:law_organic_regulation
					government_transfer_of_power = dictatorial
				}
				create_character = {
					template = WAL_barbu_dimitrie_stribei
					ruler = yes
				}
			}
		}
		resign_and_retire = yes
	}
	else_if = {
		limit = {
			has_template = WAL_barbu_dimitrie_stribei
			year >= 1853
		}
		owner ?= {
			if = {
				limit = {
					country_has_monarchy_law = yes
					country_has_non_monarchy_law = no
					is_player = no
					exists = c:MOL
					c:MOL = {
						is_player = no
						country_has_monarchy_law = yes
						country_has_non_monarchy_law = no
					}
					trigger_if = {
						limit = {
							is_subject = yes
						}
						c:MOL = { is_subject = yes }
						top_overlord = {
							THIS = c:MOL.top_overlord
						}
					}
					NOR = {
						is_subject_of = c:MOL
						c:MOL = {
							is_subject_of = c:WAL
						}
					}
				}
				set_variable = romania_union_partner
				c:MOL = {
					set_variable = {
						name = romanian_union_var
						value = 0
					}
					set_variable = {
						name = ROM_united_principality
					}
					create_diplomatic_pact = {
						country = c:WAL
						type = personal_union
					}
					add_journal_entry = { type = je_all_for_one }
				}
			}
		}
	}
}
