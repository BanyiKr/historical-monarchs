	effect = {
		# Monarch initialization
		scope:target = {
			if = {
				limit = {
					country_has_monarchy_law = yes
					country_has_non_monarchy_law = no # To handle an execution order issue
					trigger_if = {
						limit = {
							exists = ruler
						}
						ruler = {
							is_historical = no
							is_monarch = yes
						}
					}
					trigger_if = {
						limit = {
							is_subject = yes
						}
						NOT = {
							is_subject_type = subject_type_personal_union
						}
					}
				}
				country_initialize_monarch_effect = yes
			}
		}
		### Great Britain fracture - if Scotland becomes independent, revert to Egnland
		if = {
			limit = {
				root = {
					c:GBR ?= this
				}
				scope:target = {
					c:SCO ?= this
				}
			}
			change_tag = ENG
		}
	}
}
