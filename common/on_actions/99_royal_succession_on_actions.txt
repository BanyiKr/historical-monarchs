# Root = Character
historical_successions_on_character_death = {
	effect = {
		if = {
			limit = {
				is_ruler_of_own_country = yes
				owner ?= {
					has_gov_regency = no # Assigning a regent should never fire this on_action
					government_transfer_of_power = dictatorial
				}
			}
			if = { ### Dalai Lamas
				limit = {
					owner ?= {
						has_law_or_variant = law_type:law_theocracy
						OR = {
							c:TIB ?= this
							was_formed_from = TIB
						}
					}
				}
				character_tibet_ruler_succession_chain_effect = yes
			}
			else_if = { ### Popes
				limit = {
					owner ?= {
						has_law_or_variant = law_type:law_theocracy
						OR = {
							c:PAP ?= this
							was_formed_from = PAP
						}
					}
				}
				character_papal_states_ruler_succession_chain_effect = yes
			}
			else_if = { ### Bhutanese Druk Desis
				limit = {
					owner ?= {
						has_law_or_variant = law_type:law_state_religion
						OR = {
							c:BHU ?= this
							was_formed_from = BHU
						}
					}
				}
				character_bhutan_druk_desi_succession_chain_effect = yes
			}
			else_if = { # Hungarian palatines
				limit = {
					owner ?= {
						is_subject_type = subject_type_crown_land
						c:HUN ?= this
						c:AUS ?= {
							has_law = law_type:law_monarchy
						}
						is_subject_of = c:AUS
						has_gov_regency = no
						is_domain_alliance_gov = no
					}
				}
				character_hungary_palatine_succession_chain_effect = yes
			} 
			else_if = { # Transylvanian governors
				limit = {
					owner ?= {
						c:TRS ?= this
						is_subject_type = subject_type_crown_land
						overlord ?= {
							has_law = law_type:law_monarchy
						}
						has_gov_regency = no
						is_domain_alliance_gov = no

# Root = Releasing Country
# scope:target = Released Country
historical_monarchs_new_country_on_action = {
	effect = {
		# Monarch initialization
		scope:target = {
			if = {
				limit = {
					country_has_monarchy_law = yes
					country_has_non_monarchy_law = no # To handle an execution order issue
					trigger_if = {
						limit = {
							exists = ruler
						}
						ruler = {
							is_historical = no
							is_monarch = yes
						}
					}
					trigger_if = {
						limit = {
							is_subject = yes
						}
						NOT = {
							is_subject_type = subject_type_personal_union
						}
					}
				}
				country_initialize_monarch_effect = yes
			}
		}
		### Great Britain fracture - if Scotland becomes independent, revert to Egnland
		if = {
			limit = {
				root = {
					c:GBR ?= this
				}
				scope:target = {
					c:SCO ?= this
				}
			}
			change_tag = ENG
		}
	}
}
